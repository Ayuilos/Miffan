cmake_minimum_required(VERSION 3.22.1)
project(miffanstream LANGUAGES C CXX)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../..")
set(SOURCES "${ROOT}/stream/build/sources")
set(SSL "${ROOT}/rdp/build/native/${ANDROID_ABI}")
if(NOT EXISTS "${SOURCES}/src/Limelight.h")
    message(FATAL_ERROR "Run python3 stream/scripts/prepare.py first")
endif()
set(OPENSSL_INCLUDE_DIR "${SSL}/include" CACHE PATH "" FORCE)
set(OPENSSL_CRYPTO_LIBRARY "${SSL}/lib/libcrypto.a" CACHE FILEPATH "" FORCE)
set(OPENSSL_SSL_LIBRARY "${SSL}/lib/libssl.a" CACHE FILEPATH "" FORCE)
set(BUILD_SHARED_LIBS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${SOURCES}" common)
add_library(miffanstream SHARED stream_jni.cpp)
target_compile_features(miffanstream PRIVATE cxx_std_17)
target_include_directories(miffanstream PRIVATE "${SSL}/include")
target_link_libraries(miffanstream PRIVATE moonlight-common-c "${SSL}/lib/libssl.a" "${SSL}/lib/libcrypto.a" log android)
target_link_options(miffanstream PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
